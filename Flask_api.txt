import sys
import pandas as pd
import pickle
from flask import Flask, request, jsonify
import numpy as np
import os
import time
from sklearn.base import BaseEstimator, TransformerMixin

app = Flask(__name__)

# Mapping of scenario IDs to model paths
model_mapping = {
    "pbp": {"model": "./claim_models/pbp.pkl"},
    "eop": {"model": "./claim_models/EOP.pkl"}
}

def load_model(scenario_id):
    t1 = time.perf_counter()
    model_info = model_mapping.get(scenario_id)
    if not model_info:
        raise ValueError("Invalid Scenario Name")

    model_path = model_info["model"]
    if not os.path.exists(model_path):
        raise FileNotFoundError("Model file not found!")

    try:
        with open(model_path, 'rb') as model_file:
            model_contents = pickle.load(model_file)

        if len(model_contents) == 3:
            pipeline, label_encoders, feature_data = model_contents
        else:
            feature_data = model_contents
            pipeline = None
            label_encoders = None
            print("Model and Feature Data Loaded Successfully.")

    except Exception as e:
        print(f"Error loading model: {e}")
        feature_data = None
        pipeline = None
        label_encoders = None

    t2 = time.perf_counter()
    print(f'load model diff: {t2 - t1}')
    return pipeline, label_encoders, feature_data


@app.route("/retrieve_features", methods=["POST"])
def retrieve_features():
    try:
        req_data = request.get_json()

        scenario_name = req_data.get("scenario_name")
        scenario_id = req_data.get("scenario_id")
        row_limit = req_data.get("no_of_claims")
        filters = req_data.get("filters")

        if not scenario_id or not filters:
            return jsonify({"error": "scenario_id and filters are required"}), 400

        pipeline, label_encoders, feature_data = load_model(scenario_id)

        if feature_data is None:
            return jsonify({"error": "Feature data is not available!"}), 500

        df_input = pd.DataFrame(filters)

        # Normalize column names
        df_input.columns = df_input.columns.str.strip().str.lower().str.replace(r'\s+', '_', regex=True)
        feature_data.columns = feature_data.columns.str.strip().str.lower().str.replace(r'\s+', '_', regex=True)

        categorical_columns = [col for col in df_input.columns if col in feature_data.columns]
        if not categorical_columns:
            return jsonify({"error": "No valid categorical columns found for filtering!"}), 400

        print(f"Found matching categorical columns: {categorical_columns}")

        unique_combinations = df_input[categorical_columns].drop_duplicates()
        final_filtered_data = []

        for _, row in unique_combinations.iterrows():
            filter_condition = pd.Series(True, index=feature_data.index)

            for col in categorical_columns:
                # Debug prints
                print(f"\nComparing columns: '{col}'")
                print(f"Row value: {repr(row[col])}")
                print(f"Feature column dtype: {feature_data[col].dtype}")
                print(f"Row value dtype: {type(row[col])}")
                print(f"Unique values in feature_data[{col}]: {feature_data[col].unique()}")

                comparison_result = feature_data[col] == row[col]
                print(f"Matches found in col '{col}': {comparison_result.sum()}")

                # Apply filter
                filter_condition &= comparison_result

            filtered_subset = feature_data[filter_condition]
            if not filtered_subset.empty:
                filtered_subset = filtered_subset.head(row_limit)
                final_filtered_data.append(filtered_subset)

        if final_filtered_data:
            status = "Success"
            final_filtered_data = pd.concat(final_filtered_data, ignore_index=True)
            cleaned_final_data = final_filtered_data.replace({np.nan: None, np.inf: None, -np.inf: None})
            json_data = cleaned_final_data.to_dict(orient="records")
            return jsonify({
                "Scenario_name": scenario_name,
                "Scenario_id": scenario_id,
                "Status": status,
                "Claims": json_data
            })
        else:
            return jsonify({
                "Scenario_name": scenario_name,
                "Scenario_id": scenario_id,
                "Status": "Data not found"
            })

    except Exception as e:
        return jsonify({"error": str(e)}), 500


if __name__ == "__main__":
    app.run(host="0.0.0.0", port=8000, debug=True)
